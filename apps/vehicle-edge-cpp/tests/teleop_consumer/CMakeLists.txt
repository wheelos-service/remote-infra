cmake_minimum_required(VERSION 3.16)
project(vehicle_teleop_consumer LANGUAGES CXX)
find_package(VehicleAuthSDK 0.1 CONFIG REQUIRED)
find_package(VehicleTeleopSDK 0.1 CONFIG REQUIRED)
if(TARGET VehicleEdge::data OR TARGET VehicleEdge::runtime)
  message(FATAL_ERROR "Standalone teleop must not import operations targets")
endif()
add_executable(teleop_consumer main.cc)
target_link_libraries(teleop_consumer PRIVATE VehicleEdge::auth VehicleEdge::teleop)
option(CONSUMER_CAMERA_GST "Verify optional installed camera adapter" OFF)
if(CONSUMER_CAMERA_GST)
  if(NOT TARGET VehicleEdge::camera_gst)
    message(FATAL_ERROR "Installed teleop package lacks camera_gst adapter")
  endif()
  target_link_libraries(teleop_consumer PRIVATE VehicleEdge::camera_gst)
  target_compile_definitions(teleop_consumer PRIVATE CONSUMER_CAMERA_GST)
endif()
enable_testing()
add_test(NAME installed_teleop_consumer COMMAND teleop_consumer)
